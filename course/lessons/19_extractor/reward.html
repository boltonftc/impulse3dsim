🔓 1. New file: Extractor.java⩢ Copy
package org.firstinspires.ftc.teamcode;

import com.qualcomm.robotcore.hardware.HardwareMap;
import com.qualcomm.robotcore.hardware.Servo;

/*
 * Extractor -- the pollen extractor arm. A positional servo with just TWO places to be:
 *   STOWED   = up, tucked in (where it starts every match)
 *   DEPLOYED = lowered in front of the robot
 * The driver decides when: D-pad down deploys, D-pad up stows.
 */
public class Extractor implements Subsystem {

    /** The two places the arm can be. public so TeleOp can show it on telemetry. */
    public enum State {
        STOWED,     // up, tucked in -- where it starts
        DEPLOYED    // lowered in front of the robot
    }

    // Servo numbers (0.0 .. 1.0) for each state. FIND YOURS with Extractor Calibrate!
    private static final double STOWED_POS   = 0.10;
    private static final double DEPLOYED_POS = 0.75;

    private final Servo servo;                 // null if this robot has no "extractor" configured
    private State state = State.STOWED;        // where we WANT the arm to be

    public Extractor(HardwareMap hardwareMap) {
        Servo found = null;
        try {
            found = hardwareMap.get(Servo.class, "extractor");
        } catch (IllegalArgumentException e) {
            found = null;   // no "extractor" in the configuration -- run without it
        }
        servo = found;
        if (servo != null) {
            servo.setPosition(STOWED_POS);   // moves as soon as INIT runs -- hands clear!
        }
    }

    /** Lower the arm in front of the robot. */
    public void deploy() {
        state = State.DEPLOYED;
    }

    /** Raise the arm back up, out of the way. */
    public void stow() {
        state = State.STOWED;
    }

    /** Where we're sending the arm right now. */
    public State getState() {
        return state;
    }

    public boolean isDeployed() {
        return state == State.DEPLOYED;
    }

    @Override
    public void update() {
        if (servo == null) {
            return;                              // no extractor on this robot
        }
        switch (state) {
            case STOWED:
                servo.setPosition(STOWED_POS);
                break;
            case DEPLOYED:
                servo.setPosition(DEPLOYED_POS);
                break;
        }
    }
}

4. New file (a tool): ExtractorCalibrate.java⩢ Copy
package org.firstinspires.ftc.teamcode;

import com.qualcomm.robotcore.eventloop.opmode.LinearOpMode;
import com.qualcomm.robotcore.eventloop.opmode.TeleOp;
import com.qualcomm.robotcore.hardware.Servo;

/*
 * Extractor Calibrate -- move the arm with the right stick and read its servo number on the
 * Driver Hub. Write down the numbers for "up" and "down" and put them in Extractor.java.
 */
@TeleOp(name = "Extractor Calibrate", group = "Tools")
public class ExtractorCalibrate extends LinearOpMode {

    @Override
    public void runOpMode() {
        Servo servo = hardwareMap.get(Servo.class, "extractor");
        double position = 0.5;

        telemetry.addLine("Hands clear: the arm moves to the middle at START.");
        telemetry.addLine("Then move it slowly with the right stick.");
        telemetry.update();
        waitForStart();

        while (opModeIsActive()) {
            position -= gamepad1.right_stick_y * 0.003;                // stick up = bigger number
            position = Math.max(0.0, Math.min(1.0, position));        // stay between 0 and 1
            servo.setPosition(position);
            telemetry.addData("Extractor position", "%.3f", position);
            telemetry.addLine("Write down the number for UP and for DOWN.");
            telemetry.update();
        }
    }
}
Make it yours
  • Configuration: the servo must be named extractor in the Control Hub configuration (type: Servo).
  • Calibrate first. Replace 0.10 and 0.75 with the numbers you measured.
  • Moves the wrong way? Add servo.setDirection(Servo.Direction.REVERSE); in the constructor (before the first setPosition), or swap your two numbers.
  • Rather have one button? Make LB a toggle. Because it flips, it needs edge detection, like the intake's A button: if (gamepad1.left_bumper && !previousLB) { if (robot.extractor.isDeployed()) robot.extractor.stow(); else robot.extractor.deploy(); } then previousLB = gamepad1.left_bumper; (and a private boolean previousLB = false; field).
  • Two drivers? Use gamepad2.dpad_down / gamepad2.dpad_up so the second driver runs the arm.
  • Safety lock: if the arm could hit something (for example the intake running in reverse), call robot.extractor.stow() automatically when that happens.
