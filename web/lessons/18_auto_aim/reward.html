package org.firstinspires.ftc.teamcode;

import com.qualcomm.robotcore.hardware.HardwareMap;
import com.qualcomm.robotcore.hardware.PwmControl;
import com.qualcomm.robotcore.hardware.ServoImplEx;

/*
 * Turret -- an Axon MAX servo (servo mode) through a 1:3.2 overdrive into a 2.75:1 turret.
 * 350 deg x 3.2 / 2.75 = ~407 deg of turret across the servo's FULL signal (500..2500 us).
 * We ask the Control Hub for that full signal, then allow +/-200 deg: more than a full circle,
 * so the turret can point anywhere, even straight behind the robot.
 */
public class Turret implements Subsystem {

    private static final double DEGREES_PER_FULL_RANGE = 407.0;  // turret degrees across position 0..1
    public  static final double MAX_ANGLE  = 200.0;  // farthest the turret may turn each way (check the wires!)
    private static final double CENTER     = 0.5;    // servo position that points the turret forward
    private static final double MIN_POS    = CENTER - MAX_ANGLE / DEGREES_PER_FULL_RANGE;
    private static final double MAX_POS    = CENTER + MAX_ANGLE / DEGREES_PER_FULL_RANGE;
    private static final double NUDGE_STEP = 0.002;  // position change per loop while a D-pad button is held

    private final ServoImplEx servo;   // null if this robot has no "turret" configured
    private double position = CENTER;

    public Turret(HardwareMap hardwareMap) {
        ServoImplEx found = null;
        try {
            found = hardwareMap.get(ServoImplEx.class, "turret");
        } catch (IllegalArgumentException e) {
            found = null;   // no "turret" in the configuration -- run without it instead of crashing
        }
        servo = found;
        if (servo != null) {
            servo.setPwmRange(new PwmControl.PwmRange(500, 2500));   // the servo's full travel, not the hub's default
            servo.setPosition(position);   // the turret moves to center as soon as INIT runs -- hands clear!
        }
    }

    /** False on a robot without a turret: everything else still works, the turret just does nothing. */
    public boolean hasTurret() {
        return servo != null;
    }

    /** Swing a little to the left (counter-clockwise). For testing by hand. */
    public void nudgeLeft() {
        setPosition(position + NUDGE_STEP);
    }

    /** Swing a little to the right (clockwise). For testing by hand. */
    public void nudgeRight() {
        setPosition(position - NUDGE_STEP);
    }

    /** Point at an angle: 0 = forward, + = left. Auto-aim calls this every loop. */
    public void setAngle(double degrees) {
        setPosition(CENTER + degrees / DEGREES_PER_FULL_RANGE);
    }

    /** The angle we're aiming at, in degrees (0 = forward, + = left). */
    public double getAngle() {
        return (position - CENTER) * DEGREES_PER_FULL_RANGE;
    }

    private void setPosition(double p) {
        position = Math.max(MIN_POS, Math.min(MAX_POS, p));   // never past the safe limits
    }

    @Override
    public void update() {
        if (servo != null) {
            servo.setPosition(position);
        }
    }
}
2. New file: AutoAim.java⩢ Copy
package org.firstinspires.ftc.teamcode;

/*
 * AutoAim -- keeps the turret pointed at a hive cell, using where odometry says we are.
 *
 * Field coordinates (seen from YOUR drivers, red or blue): (0, 0) is the field center, +X is right,
 * +Y is away from the drivers, distances in inches. Angles: 0 = +X, counter-clockwise is +.
 *
 *     target cell  = the near hive's cell on the same side of the field as the robot
 *     turret angle = field angle to the cell - robot heading + trim
 */
public class AutoAim {

    // Where the NEAR hive's two cells are (the hive in front of your drivers), in field inches.
    // Same numbers on red or blue. MEASURE these on your field!
    private static final double LEFT_CELL_X  = -11.4;
    private static final double RIGHT_CELL_X =  10.2;
    private static final double CELL_Y       = -12.75;

    private static final double DEAD_ZONE  = 6.0;    // inches each side of x = 0 where we keep the current cell
    private static final double TRIM_MAX   = 20.0;   // the driver's trim can shift the aim at most this far
    private static final double TRIM_SPEED = 0.5;    // degrees of trim per loop with the stick pushed all the way

    private boolean aimingRight = true;              // the right cell is raised at the start of the match
    private double trimDeg = 0.0;
    private String status = "position unknown";

    /** Shift the aim with a stick: push right to aim further right. The trim stays where you leave it. */
    public void trim(double stick) {
        trimDeg -= stick * TRIM_SPEED;                              // right on the stick = clockwise = minus
        trimDeg = Math.max(-TRIM_MAX, Math.min(TRIM_MAX, trimDeg));
    }

    /** Call every loop: picks the cell, works out the turret angle, and aims. */
    public void aim(Localization localization, Turret turret) {
        if (!turret.hasTurret()) {
            status = "no turret on this robot";
            return;
        }
        // Don't guess! Until we know where we are, point straight ahead and wait.
        if (!localization.isPoseKnown()) {
            turret.setAngle(0);
            status = "position unknown: park at HOME and hold RB";
            return;
        }

        double robotX = localization.getXInches();
        double robotY = localization.getYInches();

        // 0. Which cell? The one on our side. Inside the dead zone, keep the one we had.
        if (robotX > DEAD_ZONE)  aimingRight = true;
        if (robotX < -DEAD_ZONE) aimingRight = false;
        double targetX = aimingRight ? RIGHT_CELL_X : LEFT_CELL_X;
        double targetY = CELL_Y;

        // 1. The arrow from the robot to the cell: how far over, how far up.
        double dx = targetX - robotX;
        double dy = targetY - robotY;

        // 2. Which way does that arrow point on the field?
        double fieldAngle = Math.toDegrees(Math.atan2(dy, dx));

        // 3. Measure it from the robot's front instead, and add the driver's trim.
        double turretAngle = fieldAngle - localization.getHeadingDegrees() + trimDeg;

        // 4. Keep it between -180 and 180 (340 deg and -20 deg point the same way).
        turretAngle = wrapDegrees(turretAngle);

        // 5. Near the back there are TWO ways to point the same direction (like +190 or -170).
        //    Use the other way if the turret can reach it AND it's a shorter swing from where we are now.
        double otherWay = (turretAngle > 0) ? turretAngle - 360 : turretAngle + 360;
        double now = turret.getAngle();
        if (Math.abs(otherWay) <= Turret.MAX_ANGLE
                && Math.abs(otherWay - now) < Math.abs(turretAngle - now)) {
            turretAngle = otherWay;
        }

        status = String.format("%s cell, turret %.0f deg", aimingRight ? "RIGHT" : "LEFT", turretAngle);
        turret.setAngle(turretAngle);
    }

    private static double wrapDegrees(double degrees) {
        while (degrees > 180)   degrees -= 360;
        while (degrees <= -180) degrees += 360;
        return degrees;
    }

    public String getStatus() { return status; }
    public double getTrim()   { return trimDeg; }
}
3. In Localization.java, with the other methods⩢ Copy
    // HOME: a spot the drivers can find by feel. Back of the robot against YOUR drivers' wall,
    // facing away from it, left side on the loading zone's right-hand tape line.
    // These numbers come from the simulator -- MEASURE your field and your robot!
    private static final double ROBOT_WIDTH  = 18.0;   // inches, side to side
    private static final double ROBOT_LENGTH = 18.0;   // inches, front to back
    private static final double TAPE_X = -22.0;        // the loading zone's right-hand tape line
    private static final double WALL_Y = -71.0;        // the inside face of the drivers' wall
    private static final double HOME_X = TAPE_X + ROBOT_WIDTH / 2;    // -13: the robot's CENTER, not its side
    private static final double HOME_Y = WALL_Y + ROBOT_LENGTH / 2;   // -62
    private static final double HOME_HEADING = 90.0;                  // facing away from the drivers

    private boolean poseKnown = false;   // stays false until auto hands us a pose, or the driver resets at HOME

    /** Tell the Pinpoint where the robot is right now: field inches, heading in degrees. */
    public void setPose(double xInches, double yInches, double headingDegrees) {
        latestPose = new Pose2D(DistanceUnit.INCH, xInches, yInches, AngleUnit.DEGREES, headingDegrees);
        if (pinpoint != null) {
            pinpoint.setPosition(latestPose);
        }
        poseKnown = true;
    }

    /** The robot is parked at HOME: that's exactly where we are. */
    public void setPoseToHome() {
        setPose(HOME_X, HOME_Y, HOME_HEADING);
    }

    /** True once we know where we are on the field (from auto, or a HOME reset). */
    public boolean isPoseKnown() {
        return poseKnown;
    }

9. Optional: aim during auto too

Path Capture autos aim on their own: when your package has AutoAim.java, Path Capture adds these lines for you. For a hand-written auto (like SimpleAuto), add them yourself. First, right after follower.setStartingPose(startPose);, tell Localization the same start pose so it knows where you are:

        robot.localization.setPose(startPose.getX(), startPose.getY(), Math.toDegrees(startPose.getHeading()));
        AutoAim aim = new AutoAim();

Then, inside every follow loop (and once just before you shoot), point the turret:

            aim.aim(robot.localization, robot.turret);

The turret servo holds its angle, so while the robot sits still and shoots, it stays on the hive.

Make it yours
  • Customized lesson 17's Turret? If you added setDirection(REVERSE) or changed CENTER, put those back into the new file.
  • Measure the turret. Use setAngle(90) and setAngle(180) and check with a protractor. If they're off, adjust DEGREES_PER_FULL_RANGE.
  • Wires not ready for ±200°? Lower MAX_ANGLE (but keep it above 180, or some directions can't be reached).
  • Measure the field. The cell positions and the HOME spot come from the simulator's field model. Check them on a real field with a tape measure, from the field center.
  • Measure your robot. Set ROBOT_WIDTH and ROBOT_LENGTH to your robot's real size (to the outside of the bumpers), so HOME lands on the robot's center.
  • A different HOME? Any spot works if the drivers can find it by feel. Change TAPE_X, WALL_Y and HOME_HEADING to match. If the robot doesn't face away from the drivers there, don't reset the IMU on that button.
  • Trim too slow or too fast? Change TRIM_SPEED. Need more room? Raise TRIM_MAX.
  • Target switches too eagerly, or not soon enough? Change DEAD_ZONE. Wider means more driving before it switches.
  • Blue alliance? Nothing changes. Every number is measured from your own drivers' wall, and the field looks the same from both sides, so blue uses the same code: aim at the near hive, HOME is the mirror-image spot by your own loading zone, and autos start on the mirror-image spot.
